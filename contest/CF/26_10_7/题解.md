B题
题意：给一长为n的字符串，和n份文档，对于$s_i$,如果 $s_i = 1$，则将文档i放到桌子最顶端；如果$s_i = 2$,如果此时桌子上有文档则打印最顶端的文档，否则打印文档i；如果$s_i = 3$,打印文档i。

做法：这题其涉及到一个后进先出的思想，也就是栈。
这里用数组模拟栈。新开两个数组$st$和$vis$，遍历$i$，如果$s_i = 1$，把$i$加入到数组$st$中；如果$s_i = 2$，如果数组st非空，取出数组$st$中最后的元素$j$，把数组$vis$下标$j$标记为1，把数组$st$最后的元素删掉，否则把数组$vis$下标$i$标记为1；如果 $s_i = 3$，把数组vis下标i标记为1。最后开一个答案数组，再次遍历$i$，如果$vis[i]==0$，把$i$加入答案数组
代码

```cpp    
#include<bits/stdc++.h>
using namespace std;
using vt = vector<int>;

void solve()
{
    int n;
    cin >> n;
    string s;
    cin >> s;
    vt st;
    vt vis(n);
    for (int i = 0;i < n;i++)
    {
        if (s[i] == '1') st.push_back(i);
        else if (s[i] == '2') 
        {
            if (st.size()) 
            {
                vis[st.top()] = 1;
                st.pop();
            }
            else 
            {
                vis[i] = 1;
            }
        }
        else 
        {
            vis[i] = 1;
        }
    }
    vt ans;
    for (int i = 0;i < n;i++)
    {
        if (!vis[i]) ans.push_back(i + 1);
    }
    cout << ans.size() << endl;
    for (int x : ans) cout << x << " ";
    if (ans.size()) cout << endl;
}

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cout << fixed << setprecision(15);
    int _ = 1;
    cin >> _;
    while (_ --) solve();

    return 0;
}
```
